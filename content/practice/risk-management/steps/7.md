An interview-sized engine. Rules are stateless strategies, the velocity store does atomic record-and-count, and config is an immutable snapshot swapped atomically.

```java
import java.time.*;
import java.util.*;
import java.util.concurrent.ConcurrentHashMap;
import java.util.concurrent.CopyOnWriteArrayList;
import java.util.concurrent.atomic.AtomicReference;

enum Decision { ALLOW, REVIEW, BLOCK }       // ordered by severity

record Transaction(String id, String userId, long amountMinor, String currency,
                   String country, String deviceId, String ip, Instant at) {}

record UserProfile(String userId, String homeCountry, Set<String> knownDevices) {}

record RuleResult(String ruleId, int score, Decision verdict, String reason) {
    static RuleResult pass(String ruleId) { return new RuleResult(ruleId, 0, Decision.ALLOW, null); }
    boolean triggered() { return score > 0 || verdict != Decision.ALLOW; }
}

record Assessment(String txnId, Decision decision, int score, List<RuleResult> hits) {}

interface Rule {                              // Strategy: one check, one class
    String id();
    RuleResult evaluate(Transaction t, UserProfile p);
}

interface DecisionListener { void onDecision(Assessment a); }   // audit log, review queue, alerts

/** Score thresholds → decision. Swappable per merchant or market. */
record ThresholdPolicy(int reviewAt, int blockAt) {
    Decision decide(int score) {
        return score >= blockAt ? Decision.BLOCK : score >= reviewAt ? Decision.REVIEW : Decision.ALLOW;
    }
}

class LargeAmountRule implements Rule {
    private final long limitMinor; private final int score;
    LargeAmountRule(long limitMinor, int score) { this.limitMinor = limitMinor; this.score = score; }
    public String id() { return "large-amount"; }
    public RuleResult evaluate(Transaction t, UserProfile p) {
        return t.amountMinor() > limitMinor
            ? new RuleResult(id(), score, Decision.ALLOW, "amount " + t.amountMinor() + " > " + limitMinor)
            : RuleResult.pass(id());
    }
}

class BlocklistRule implements Rule {
    private final Set<String> blocked;        // "user:u1", "device:d9", "ip:1.2.3.4"
    BlocklistRule(Set<String> blocked) { this.blocked = blocked; }
    public String id() { return "blocklist"; }
    public RuleResult evaluate(Transaction t, UserProfile p) {
        for (String key : List.of("user:" + t.userId(), "device:" + t.deviceId(), "ip:" + t.ip()))
            if (blocked.contains(key)) return new RuleResult(id(), 0, Decision.BLOCK, key + " is blocklisted");
        return RuleResult.pass(id());
    }
}

class NewDeviceRule implements Rule {
    private final int score;
    NewDeviceRule(int score) { this.score = score; }
    public String id() { return "new-device"; }
    public RuleResult evaluate(Transaction t, UserProfile p) {
        return p.knownDevices().contains(t.deviceId())
            ? RuleResult.pass(id())
            : new RuleResult(id(), score, Decision.ALLOW, "unknown device " + t.deviceId());
    }
}

class CountryMismatchRule implements Rule {
    private final int score;
    CountryMismatchRule(int score) { this.score = score; }
    public String id() { return "country-mismatch"; }
    public RuleResult evaluate(Transaction t, UserProfile p) {
        return t.country().equals(p.homeCountry())
            ? RuleResult.pass(id())
            : new RuleResult(id(), score, Decision.ALLOW, t.country() + " != home " + p.homeCountry());
    }
}

/** Sliding-window event log per key. recordAndCount is atomic per key. */
class VelocityStore {
    private final ConcurrentHashMap<String, ArrayDeque<Instant>> events = new ConcurrentHashMap<>();

    int recordAndCount(String key, Instant at, Duration window) {
        int[] n = new int[1];
        events.compute(key, (k, q) -> {
            if (q == null) q = new ArrayDeque<>();
            q.addLast(at);
            Instant cutoff = at.minus(window);
            while (!q.isEmpty() && !q.peekFirst().isAfter(cutoff)) q.pollFirst();
            n[0] = q.size();
            return q;
        });
        return n[0];
    }
}

class VelocityRule implements Rule {
    private final VelocityStore store; private final int maxCount; private final Duration window; private final int score;
    VelocityRule(VelocityStore store, int maxCount, Duration window, int score) {
        this.store = store; this.maxCount = maxCount; this.window = window; this.score = score;
    }
    public String id() { return "velocity"; }
    public RuleResult evaluate(Transaction t, UserProfile p) {
        int n = store.recordAndCount(id() + ":" + t.userId(), t.at(), window);
        return n > maxCount
            ? new RuleResult(id(), score, Decision.ALLOW, n + " txns in " + window)
            : RuleResult.pass(id());
    }
}

record RuleSet(List<Rule> rules, ThresholdPolicy policy) {
    RuleSet { rules = List.copyOf(rules); }
}

class RiskEngine {
    private final AtomicReference<RuleSet> config;        // immutable snapshot, hot-swappable
    private final List<DecisionListener> listeners = new CopyOnWriteArrayList<>();

    RiskEngine(List<Rule> rules, ThresholdPolicy policy) {
        config = new AtomicReference<>(new RuleSet(rules, policy));
    }

    void reload(List<Rule> rules, ThresholdPolicy policy) {
        config.set(new RuleSet(rules, policy));          // one atomic swap
    }

    void subscribe(DecisionListener l) { listeners.add(l); }

    Assessment assess(Transaction t, UserProfile p) {
        List<RuleResult> hits = new ArrayList<>();
        int score = 0;
        Decision hard = Decision.ALLOW;
        RuleSet cfg = config.get();                       // read once
        for (Rule r : cfg.rules()) {
            RuleResult res;
            try {
                res = r.evaluate(t, p);
            } catch (RuntimeException e) {            // fail safe: a broken rule sends to review
                res = new RuleResult(r.id(), 0, Decision.REVIEW, "rule error: " + e.getMessage());
            }
            if (!res.triggered()) continue;
            hits.add(res);
            score += res.score();
            if (res.verdict().compareTo(hard) > 0) hard = res.verdict();
            if (hard == Decision.BLOCK) break;         // short-circuit
        }
        Decision byScore = cfg.policy().decide(score);
        Decision decision = byScore.compareTo(hard) > 0 ? byScore : hard;
        Assessment a = new Assessment(t.id(), decision, score, List.copyOf(hits));
        for (DecisionListener l : listeners) l.onDecision(a);
        return a;
    }
}
```

```go
package risk

import (
	"fmt"
	"sync"
	"sync/atomic"
	"time"
)

// Decision is ordered by severity.
type Decision int

const (
	Allow Decision = iota
	Review
	Block
)

type Transaction struct {
	ID, UserID, Currency, Country, DeviceID, IP string
	AmountMinor                                 int64
	At                                          time.Time
}

type UserProfile struct {
	UserID       string
	HomeCountry  string
	KnownDevices map[string]bool
}

type RuleResult struct {
	RuleID  string
	Score   int
	Verdict Decision
	Reason  string
}

func (r RuleResult) Triggered() bool { return r.Score > 0 || r.Verdict != Allow }

type Assessment struct {
	TxnID    string
	Decision Decision
	Score    int
	Hits     []RuleResult
}

// Rule is the Strategy: one check per implementation.
type Rule interface {
	ID() string
	Evaluate(t Transaction, p UserProfile) (RuleResult, error)
}

// Listener receives every decision: audit log, review queue, alerts.
type Listener func(Assessment)

// ThresholdPolicy maps a total score to a decision.
type ThresholdPolicy struct{ ReviewAt, BlockAt int }

func (tp ThresholdPolicy) Decide(score int) Decision {
	switch {
	case score >= tp.BlockAt:
		return Block
	case score >= tp.ReviewAt:
		return Review
	}
	return Allow
}

type LargeAmountRule struct {
	LimitMinor int64
	Score      int
}

func (LargeAmountRule) ID() string { return "large-amount" }
func (r LargeAmountRule) Evaluate(t Transaction, _ UserProfile) (RuleResult, error) {
	if t.AmountMinor > r.LimitMinor {
		return RuleResult{r.ID(), r.Score, Allow, fmt.Sprintf("amount %d > %d", t.AmountMinor, r.LimitMinor)}, nil
	}
	return RuleResult{RuleID: r.ID()}, nil
}

// BlocklistRule blocks on keys such as "user:u1", "device:d9", "ip:1.2.3.4".
type BlocklistRule struct{ Blocked map[string]bool }

func (BlocklistRule) ID() string { return "blocklist" }
func (r BlocklistRule) Evaluate(t Transaction, _ UserProfile) (RuleResult, error) {
	for _, k := range []string{"user:" + t.UserID, "device:" + t.DeviceID, "ip:" + t.IP} {
		if r.Blocked[k] {
			return RuleResult{r.ID(), 0, Block, k + " is blocklisted"}, nil
		}
	}
	return RuleResult{RuleID: r.ID()}, nil
}

type NewDeviceRule struct{ Score int }

func (NewDeviceRule) ID() string { return "new-device" }
func (r NewDeviceRule) Evaluate(t Transaction, p UserProfile) (RuleResult, error) {
	if p.KnownDevices[t.DeviceID] {
		return RuleResult{RuleID: r.ID()}, nil
	}
	return RuleResult{r.ID(), r.Score, Allow, "unknown device " + t.DeviceID}, nil
}

type CountryMismatchRule struct{ Score int }

func (CountryMismatchRule) ID() string { return "country-mismatch" }
func (r CountryMismatchRule) Evaluate(t Transaction, p UserProfile) (RuleResult, error) {
	if t.Country == p.HomeCountry {
		return RuleResult{RuleID: r.ID()}, nil
	}
	return RuleResult{r.ID(), r.Score, Allow, t.Country + " != home " + p.HomeCountry}, nil
}

// VelocityStore is a sliding-window event log per key.
type VelocityStore struct {
	mu     sync.Mutex
	events map[string][]time.Time
}

func NewVelocityStore() *VelocityStore { return &VelocityStore{events: map[string][]time.Time{}} }

// RecordAndCount appends an event and returns how many fall in (at-window, at], atomically.
func (s *VelocityStore) RecordAndCount(key string, at time.Time, window time.Duration) int {
	s.mu.Lock()
	defer s.mu.Unlock()
	q := append(s.events[key], at)
	cutoff := at.Add(-window)
	i := 0
	for i < len(q) && !q[i].After(cutoff) {
		i++
	}
	q = q[i:]
	s.events[key] = q
	return len(q)
}

type VelocityRule struct {
	Store    *VelocityStore
	MaxCount int
	Window   time.Duration
	Score    int
}

func (VelocityRule) ID() string { return "velocity" }
func (r VelocityRule) Evaluate(t Transaction, _ UserProfile) (RuleResult, error) {
	n := r.Store.RecordAndCount(r.ID()+":"+t.UserID, t.At, r.Window)
	if n > r.MaxCount {
		return RuleResult{r.ID(), r.Score, Allow, fmt.Sprintf("%d txns in %s", n, r.Window)}, nil
	}
	return RuleResult{RuleID: r.ID()}, nil
}

type config struct {
	rules  []Rule
	policy ThresholdPolicy
}

type Engine struct {
	cfg       atomic.Pointer[config] // immutable snapshot, hot-swappable
	mu        sync.RWMutex
	listeners []Listener
}

func NewEngine(rules []Rule, policy ThresholdPolicy) *Engine {
	e := &Engine{}
	e.Reload(rules, policy)
	return e
}

func (e *Engine) Reload(rules []Rule, policy ThresholdPolicy) {
	e.cfg.Store(&config{rules: append([]Rule(nil), rules...), policy: policy})
}

func (e *Engine) Subscribe(l Listener) {
	e.mu.Lock()
	defer e.mu.Unlock()
	e.listeners = append(e.listeners, l)
}

func (e *Engine) Assess(t Transaction, p UserProfile) Assessment {
	cfg := e.cfg.Load()
	var hits []RuleResult
	score, hard := 0, Allow
	for _, r := range cfg.rules {
		res, err := r.Evaluate(t, p)
		if err != nil { // fail safe: a broken rule sends to review
			res = RuleResult{r.ID(), 0, Review, "rule error: " + err.Error()}
		}
		if !res.Triggered() {
			continue
		}
		hits = append(hits, res)
		score += res.Score
		hard = max(hard, res.Verdict)
		if hard == Block {
			break // short-circuit
		}
	}
	a := Assessment{TxnID: t.ID, Decision: max(cfg.policy.Decide(score), hard), Score: score, Hits: hits}
	e.mu.RLock()
	ls := e.listeners
	e.mu.RUnlock()
	for _, l := range ls {
		l(a)
	}
	return a
}
```

```python
import threading
from abc import ABC, abstractmethod
from collections import deque
from dataclasses import dataclass, field
from datetime import datetime, timedelta
from enum import IntEnum
from typing import Callable, Dict, FrozenSet, List, Optional, Tuple


class Decision(IntEnum):  # ordered by severity
    ALLOW = 0
    REVIEW = 1
    BLOCK = 2


@dataclass(frozen=True)
class Transaction:
    id: str
    user_id: str
    amount_minor: int
    currency: str
    country: str
    device_id: str
    ip: str
    at: datetime


@dataclass(frozen=True)
class UserProfile:
    user_id: str
    home_country: str
    known_devices: FrozenSet[str] = frozenset()


@dataclass(frozen=True)
class RuleResult:
    rule_id: str
    score: int = 0
    verdict: Decision = Decision.ALLOW
    reason: Optional[str] = None

    @property
    def triggered(self) -> bool:
        return self.score > 0 or self.verdict != Decision.ALLOW


@dataclass(frozen=True)
class Assessment:
    txn_id: str
    decision: Decision
    score: int
    hits: Tuple[RuleResult, ...] = field(default_factory=tuple)


class Rule(ABC):  # Strategy: one check per class
    id = ""

    @abstractmethod
    def evaluate(self, t: Transaction, p: UserProfile) -> RuleResult: ...


@dataclass(frozen=True)
class ThresholdPolicy:
    review_at: int
    block_at: int

    def decide(self, score: int) -> Decision:
        if score >= self.block_at:
            return Decision.BLOCK
        if score >= self.review_at:
            return Decision.REVIEW
        return Decision.ALLOW


class LargeAmountRule(Rule):
    id = "large-amount"

    def __init__(self, limit_minor: int, score: int):
        self.limit_minor, self.score = limit_minor, score

    def evaluate(self, t, p):
        if t.amount_minor > self.limit_minor:
            return RuleResult(self.id, self.score, reason="amount %d > %d" % (t.amount_minor, self.limit_minor))
        return RuleResult(self.id)


class BlocklistRule(Rule):
    id = "blocklist"

    def __init__(self, blocked: FrozenSet[str]):  # "user:u1", "device:d9", "ip:1.2.3.4"
        self.blocked = blocked

    def evaluate(self, t, p):
        for key in ("user:" + t.user_id, "device:" + t.device_id, "ip:" + t.ip):
            if key in self.blocked:
                return RuleResult(self.id, 0, Decision.BLOCK, key + " is blocklisted")
        return RuleResult(self.id)


class NewDeviceRule(Rule):
    id = "new-device"

    def __init__(self, score: int):
        self.score = score

    def evaluate(self, t, p):
        if t.device_id in p.known_devices:
            return RuleResult(self.id)
        return RuleResult(self.id, self.score, reason="unknown device " + t.device_id)


class CountryMismatchRule(Rule):
    id = "country-mismatch"

    def __init__(self, score: int):
        self.score = score

    def evaluate(self, t, p):
        if t.country == p.home_country:
            return RuleResult(self.id)
        return RuleResult(self.id, self.score, reason="%s != home %s" % (t.country, p.home_country))


class VelocityStore:
    """Sliding-window event log per key; record_and_count is atomic."""

    def __init__(self):
        self._events: Dict[str, deque] = {}
        self._lock = threading.Lock()

    def record_and_count(self, key: str, at: datetime, window: timedelta) -> int:
        with self._lock:
            q = self._events.setdefault(key, deque())
            q.append(at)
            cutoff = at - window
            while q and q[0] <= cutoff:
                q.popleft()
            return len(q)


class VelocityRule(Rule):
    id = "velocity"

    def __init__(self, store: VelocityStore, max_count: int, window: timedelta, score: int):
        self.store, self.max_count, self.window, self.score = store, max_count, window, score

    def evaluate(self, t, p):
        n = self.store.record_and_count(self.id + ":" + t.user_id, t.at, self.window)
        if n > self.max_count:
            return RuleResult(self.id, self.score, reason="%d txns in %s" % (n, self.window))
        return RuleResult(self.id)


class RiskEngine:
    def __init__(self, rules: List[Rule], policy: ThresholdPolicy):
        self._config = (tuple(rules), policy)  # immutable snapshot, swapped atomically
        self._listeners: List[Callable[[Assessment], None]] = []

    def reload(self, rules: List[Rule], policy: ThresholdPolicy) -> None:
        self._config = (tuple(rules), policy)  # a single reference assignment is atomic

    def subscribe(self, listener: Callable[[Assessment], None]) -> None:
        self._listeners.append(listener)

    def assess(self, t: Transaction, p: UserProfile) -> Assessment:
        rules, policy = self._config
        hits: List[RuleResult] = []
        score, hard = 0, Decision.ALLOW
        for rule in rules:
            try:
                res = rule.evaluate(t, p)
            except Exception as e:  # fail safe: a broken rule sends to review
                res = RuleResult(rule.id, 0, Decision.REVIEW, "rule error: %s" % e)
            if not res.triggered:
                continue
            hits.append(res)
            score += res.score
            hard = max(hard, res.verdict)
            if hard == Decision.BLOCK:
                break  # short-circuit
        decision = max(policy.decide(score), hard)
        a = Assessment(t.id, Decision(decision), score, tuple(hits))
        for listener in list(self._listeners):
            listener(a)
        return a
```
