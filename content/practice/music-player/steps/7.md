A complete, interview-sized core. `PlayQueue` is pure logic. The states return the next state, including what a device switch does. `CastOutput` adapts a vendor SDK to `AudioOutput`, and `DeviceManager` tracks which devices are connected. `Player` serialises every input, device events included, under one lock.

import java.util.*;

enum RepeatMode { OFF, ONE, ALL }

record Song(String id, String title, String artist, int durationSec) {}

enum DeviceType { SPEAKER, WIRED, BLUETOOTH, CAST }

record AudioDevice(String id, String name, DeviceType type) {}

interface PlayerListener {
    void onSongChanged(Song song);
    void onStateChanged(String state);
    void onDeviceChanged(AudioDevice device);
}

/** Adapter target: the player only ever talks to this. */
interface AudioOutput {
    void start(Song song);                  // from 0:00, playing
    void prepare(Song song, int atSec);     // load at a position, paused
    void pause();
    void resume();
    void stop();                            // must be safe on a device that has gone away
    int positionSec();                      // last known position if the device is gone
}

/** A vendor SDK we can't change: URLs, milliseconds, a paused flag. */
interface CastSession {
    void loadMedia(String url, long startMs, boolean autoplay);
    void setPaused(boolean paused);
    void stopMedia();
    long currentTimeMs();
}

/** Adapter: AudioOutput → CastSession. BluetoothOutput and SpeakerOutput look the same. */
class CastOutput implements AudioOutput {
    private final CastSession session;

    CastOutput(CastSession session) { this.session = session; }

    private static String url(Song s) { return "https://cdn.example.com/tracks/" + s.id(); }

    public void start(Song s) { session.loadMedia(url(s), 0, true); }
    public void prepare(Song s, int atSec) { session.loadMedia(url(s), atSec * 1000L, false); }
    public void pause() { session.setPaused(true); }
    public void resume() { session.setPaused(false); }
    public void stop() { session.stopMedia(); }
    public int positionSec() { return (int) (session.currentTimeMs() / 1000); }
}

/** Connected devices and their adapters. Only Player calls it, under Player's lock. */
class DeviceManager {
    private final Map<String, AudioDevice> devices = new LinkedHashMap<>();
    private final Map<String, AudioOutput> outputs = new HashMap<>();
    private final String speakerId;
    private String activeId;

    DeviceManager(AudioDevice speaker, AudioOutput speakerOut) {
        speakerId = activeId = speaker.id();
        connected(speaker, speakerOut);
    }

    void connected(AudioDevice d, AudioOutput out) { devices.put(d.id(), d); outputs.put(d.id(), out); }

    void disconnected(String id) {
        if (id.equals(speakerId)) return;   // the built-in speaker never goes away
        devices.remove(id);
        outputs.remove(id);
    }

    List<AudioDevice> available() { return List.copyOf(devices.values()); }
    boolean isActive(String id) { return id.equals(activeId); }
    String speakerId() { return speakerId; }
    AudioDevice active() { return devices.get(activeId); }
    AudioOutput activeOutput() { return outputs.get(activeId); }

    Optional<AudioOutput> select(String id) {
        AudioOutput out = outputs.get(id);
        if (out != null) activeId = id;
        return Optional.ofNullable(out);
    }
}

class PlayQueue {
    private final List<Song> songs = new ArrayList<>();
    private List<Integer> order = new ArrayList<>();   // indices into songs
    private int pos = -1;
    private final Random random;

    PlayQueue(Random random) { this.random = random; }

    void load(List<Song> list, int start) {
        songs.clear();
        songs.addAll(list);
        order = new ArrayList<>();
        for (int i = 0; i < songs.size(); i++) order.add(i);
        pos = songs.isEmpty() ? -1 : start;
    }

    Optional<Song> current() {
        return pos < 0 ? Optional.empty() : Optional.of(songs.get(order.get(pos)));
    }

    /** auto = the song ended by itself; false = the user pressed Next. */
    Optional<Song> next(RepeatMode repeat, boolean auto) {
        if (pos < 0) return Optional.empty();
        if (auto && repeat == RepeatMode.ONE) return current();
        if (pos + 1 < order.size()) pos++;
        else if (repeat != RepeatMode.OFF) pos = 0;
        else return Optional.empty();
        return current();
    }

    Optional<Song> previous() {
        if (pos > 0) pos--;
        return current();
    }

    void setShuffle(boolean on) {
        if (pos < 0) return;
        int cur = order.get(pos);
        List<Integer> next = new ArrayList<>();
        for (int i = 0; i < songs.size(); i++) if (!on || i != cur) next.add(i);
        if (on) {
            Collections.shuffle(next, random);   // Fisher–Yates
            next.add(0, cur);
            pos = 0;
        } else {
            pos = cur;
        }
        order = next;
    }
}

interface PlayerState {
    PlayerState STOPPED = new Stopped(), PLAYING = new Playing(), PAUSED = new Paused();
    PlayerState play(Player p);
    PlayerState pause(Player p);
    PlayerState stop(Player p);
    PlayerState switchOutput(Player p, AudioOutput next);
    String name();
}

class Stopped implements PlayerState {
    public PlayerState play(Player p) {
        Optional<Song> s = p.queue.current();
        if (s.isEmpty()) return this;
        p.startSong(s.get());
        return PLAYING;
    }
    public PlayerState pause(Player p) { return this; }
    public PlayerState stop(Player p) { return this; }
    public PlayerState switchOutput(Player p, AudioOutput next) { p.out = next; return this; }
    public String name() { return "STOPPED"; }
}

class Playing implements PlayerState {
    public PlayerState play(Player p) { return this; }
    public PlayerState pause(Player p) { p.out.pause(); return PAUSED; }
    public PlayerState stop(Player p) { p.out.stop(); return STOPPED; }
    public PlayerState switchOutput(Player p, AudioOutput next) { p.moveOutput(next, true); return this; }
    public String name() { return "PLAYING"; }
}

class Paused implements PlayerState {
    public PlayerState play(Player p) { p.out.resume(); return PLAYING; }
    public PlayerState pause(Player p) { return this; }
    public PlayerState stop(Player p) { p.out.stop(); return STOPPED; }
    public PlayerState switchOutput(Player p, AudioOutput next) { p.moveOutput(next, false); return this; }
    public String name() { return "PAUSED"; }
}

class Player {
    final PlayQueue queue;
    final DeviceManager devices;
    AudioOutput out;                        // the active device's adapter
    private final List<PlayerListener> listeners = new ArrayList<>();
    private PlayerState state = PlayerState.STOPPED;
    private RepeatMode repeat = RepeatMode.OFF;

    Player(PlayQueue queue, DeviceManager devices) {
        this.queue = queue;
        this.devices = devices;
        this.out = devices.activeOutput();
    }

    synchronized void subscribe(PlayerListener l) { listeners.add(l); }
    synchronized void setRepeat(RepeatMode r) { repeat = r; }
    synchronized void setShuffle(boolean on) { queue.setShuffle(on); }
    synchronized String state() { return state.name(); }

    synchronized void play()  { setState(state.play(this)); }
    synchronized void pause() { setState(state.pause(this)); }
    synchronized void stop()  { setState(state.stop(this)); }
    synchronized void next()  { advance(queue.next(repeat, false)); }
    synchronized void trackEnded() { advance(queue.next(repeat, true)); }
    synchronized void previous() {
        advance(out.positionSec() > 3 ? queue.current() : queue.previous());
    }

    // ---- device selection ----
    synchronized List<AudioDevice> availableDevices() { return devices.available(); }
    synchronized void deviceConnected(AudioDevice d, AudioOutput o) { devices.connected(d, o); }
    synchronized void selectDevice(String id) { switchTo(id); }

    /** Called from the OS thread, e.g. Bluetooth dropped. Pause first, then fall back to the speaker. */
    synchronized void deviceDisconnected(String id) {
        if (devices.isActive(id)) {
            setState(state.pause(this));
            switchTo(devices.speakerId());
        }
        devices.disconnected(id);
    }

    private void switchTo(String id) {
        Optional<AudioOutput> next = devices.select(id);
        if (next.isEmpty() || next.get() == out) return;
        setState(state.switchOutput(this, next.get()));
        AudioDevice d = devices.active();
        listeners.forEach(l -> l.onDeviceChanged(d));
    }

    /** Same song, same second, on the new device. resume=false keeps it paused. */
    void moveOutput(AudioOutput next, boolean resume) {
        int at = out.positionSec();
        Song s = queue.current().orElseThrow();
        out.stop();
        out = next;
        next.prepare(s, at);
        if (resume) next.resume();
    }

    void startSong(Song s) {
        out.start(s);
        listeners.forEach(l -> l.onSongChanged(s));
    }

    private void advance(Optional<Song> s) {
        if (s.isEmpty()) { setState(state.stop(this)); return; }
        startSong(s.get());
        setState(PlayerState.PLAYING);
    }

    private void setState(PlayerState next) {
        if (next == state) return;
        state = next;
        listeners.forEach(l -> l.onStateChanged(next.name()));
    }
}

package player

import (
	"math/rand"
	"sync"
)

type RepeatMode int

const (
	RepeatOff RepeatMode = iota
	RepeatOne
	RepeatAll
)

type Song struct {
	ID, Title, Artist string
	DurationSec       int
}

type DeviceType int

const (
	Speaker DeviceType = iota
	Wired
	Bluetooth
	Cast
)

type AudioDevice struct {
	ID, Name string
	Type     DeviceType
}

type Listener interface {
	SongChanged(s Song)
	StateChanged(state string)
	DeviceChanged(d AudioDevice)
}

// AudioOutput is the Adapter target: the player only ever talks to this.
type AudioOutput interface {
	Start(s Song)              // from 0:00, playing
	Prepare(s Song, atSec int) // load at a position, paused
	Pause()
	Resume()
	Stop()            // must be safe on a device that has gone away
	PositionSec() int // last known position if the device is gone
}

// CastSession is a vendor SDK we can't change: URLs, milliseconds, a paused flag.
type CastSession interface {
	LoadMedia(url string, startMs int64, autoplay bool)
	SetPaused(paused bool)
	StopMedia()
	CurrentTimeMs() int64
}

// CastOutput adapts CastSession to AudioOutput. Bluetooth and speaker adapters look the same.
type CastOutput struct{ session CastSession }

func NewCastOutput(s CastSession) *CastOutput { return &CastOutput{session: s} }

func url(s Song) string { return "https://cdn.example.com/tracks/" + s.ID }

func (c *CastOutput) Start(s Song) { c.session.LoadMedia(url(s), 0, true) }
func (c *CastOutput) Prepare(s Song, atSec int) {
	c.session.LoadMedia(url(s), int64(atSec)*1000, false)
}
func (c *CastOutput) Pause()           { c.session.SetPaused(true) }
func (c *CastOutput) Resume()          { c.session.SetPaused(false) }
func (c *CastOutput) Stop()            { c.session.StopMedia() }
func (c *CastOutput) PositionSec() int { return int(c.session.CurrentTimeMs() / 1000) }

// DeviceManager tracks connected devices and their adapters.
// Only Player calls it, under Player's lock.
type DeviceManager struct {
	order     []string // display order
	devices   map[string]AudioDevice
	outputs   map[string]AudioOutput
	speakerID string
	activeID  string
}

func NewDeviceManager(speaker AudioDevice, out AudioOutput) *DeviceManager {
	m := &DeviceManager{
		devices:   map[string]AudioDevice{},
		outputs:   map[string]AudioOutput{},
		speakerID: speaker.ID,
		activeID:  speaker.ID,
	}
	m.Connected(speaker, out)
	return m
}

func (m *DeviceManager) Connected(d AudioDevice, out AudioOutput) {
	if _, ok := m.devices[d.ID]; !ok {
		m.order = append(m.order, d.ID)
	}
	m.devices[d.ID] = d
	m.outputs[d.ID] = out
}

func (m *DeviceManager) Disconnected(id string) {
	if id == m.speakerID { // the built-in speaker never goes away
		return
	}
	delete(m.devices, id)
	delete(m.outputs, id)
	for i, x := range m.order {
		if x == id {
			m.order = append(m.order[:i], m.order[i+1:]...)
			break
		}
	}
}

func (m *DeviceManager) Available() []AudioDevice {
	out := make([]AudioDevice, 0, len(m.order))
	for _, id := range m.order {
		out = append(out, m.devices[id])
	}
	return out
}

func (m *DeviceManager) Select(id string) (AudioOutput, bool) {
	out, ok := m.outputs[id]
	if ok {
		m.activeID = id
	}
	return out, ok
}

type Queue struct {
	songs []Song
	order []int // indices into songs
	pos   int
	rnd   *rand.Rand
}

func NewQueue(rnd *rand.Rand) *Queue { return &Queue{pos: -1, rnd: rnd} }

func (q *Queue) Load(songs []Song, start int) {
	q.songs = append([]Song(nil), songs...)
	q.order = make([]int, len(songs))
	for i := range q.order {
		q.order[i] = i
	}
	q.pos = -1
	if len(songs) > 0 {
		q.pos = start
	}
}

func (q *Queue) Current() (Song, bool) {
	if q.pos < 0 {
		return Song{}, false
	}
	return q.songs[q.order[q.pos]], true
}

// Next moves the cursor. auto is true when the song ended by itself.
func (q *Queue) Next(repeat RepeatMode, auto bool) (Song, bool) {
	switch {
	case q.pos < 0:
		return Song{}, false
	case auto && repeat == RepeatOne:
		// replay the current song
	case q.pos+1 < len(q.order):
		q.pos++
	case repeat != RepeatOff:
		q.pos = 0
	default:
		return Song{}, false
	}
	return q.Current()
}

func (q *Queue) Previous() (Song, bool) {
	if q.pos > 0 {
		q.pos--
	}
	return q.Current()
}

func (q *Queue) SetShuffle(on bool) {
	if q.pos < 0 {
		return
	}
	cur := q.order[q.pos]
	if !on {
		for i := range q.order {
			q.order[i] = i
		}
		q.pos = cur
		return
	}
	rest := make([]int, 0, len(q.songs)-1)
	for i := range q.songs {
		if i != cur {
			rest = append(rest, i)
		}
	}
	q.rnd.Shuffle(len(rest), func(i, j int) { rest[i], rest[j] = rest[j], rest[i] }) // Fisher–Yates
	q.order = append([]int{cur}, rest...)
	q.pos = 0
}

	return s
}

func (paused) stop(p *Player) state  { p.out.Stop(); return stopped{} }

	return s

type Player struct {
	mu        sync.Mutex
	queue     *Queue
	devices   *DeviceManager
	out       AudioOutput // the active device's adapter
	state     state
	repeat    RepeatMode
	listeners []Listener
}

func NewPlayer(q *Queue, devices *DeviceManager) *Player {
	out, _ := devices.Select(devices.activeID)
	return &Player{queue: q, devices: devices, out: out, state: stopped{}}
}

func (p *Player) Subscribe(l Listener) {
	p.mu.Lock()
	defer p.mu.Unlock()
	p.listeners = append(p.listeners, l)
}

func (p *Player) SetRepeat(r RepeatMode) {
	p.mu.Lock()
	defer p.mu.Unlock()
	p.repeat = r
}

func (p *Player) SetShuffle(on bool) {
	p.mu.Lock()
	defer p.mu.Unlock()
	p.queue.SetShuffle(on)
}

func (p *Player) Play()  { p.with(func() { p.setState(p.state.play(p)) }) }
func (p *Player) Pause() { p.with(func() { p.setState(p.state.pause(p)) }) }
func (p *Player) Stop()  { p.with(func() { p.setState(p.state.stop(p)) }) }
func (p *Player) Next()  { p.with(func() { p.advance(p.queue.Next(p.repeat, false)) }) }

func (p *Player) TrackEnded() { p.with(func() { p.advance(p.queue.Next(p.repeat, true)) }) }

func (p *Player) Previous() {
	p.with(func() {
		if p.out.PositionSec() > 3 {
			p.advance(p.queue.Current())
			return
		}
		p.advance(p.queue.Previous())
	})
}

// ---- device selection ----

func (p *Player) AvailableDevices() (ds []AudioDevice) {
	p.with(func() { ds = p.devices.Available() })
	return ds
}

func (p *Player) DeviceConnected(d AudioDevice, out AudioOutput) {
	p.with(func() { p.devices.Connected(d, out) })
}

func (p *Player) SelectDevice(id string) { p.with(func() { p.switchTo(id) }) }

// DeviceDisconnected is called from the OS thread, e.g. Bluetooth dropped.
// Pause first, then fall back to the speaker.
func (p *Player) DeviceDisconnected(id string) {
	p.with(func() {
		if id == p.devices.activeID {
			p.setState(p.state.pause(p))
			p.switchTo(p.devices.speakerID)
		}
		p.devices.Disconnected(id)
	})
}

func (p *Player) switchTo(id string) {
	next, ok := p.devices.Select(id)
	if !ok || next == p.out {
		return
	}
	p.setState(p.state.switchOutput(p, next))
	d := p.devices.devices[id]
	for _, l := range p.listeners {
		l.DeviceChanged(d)
	}
}

// moveOutput continues the same song at the same second on the new device.
func (p *Player) moveOutput(next AudioOutput, resume bool) {
	at := p.out.PositionSec()
	song, _ := p.queue.Current()
	p.out.Stop()
	p.out = next
	next.Prepare(song, at)
	if resume {
		next.Resume()
	}
}

func (p *Player) with(f func()) {
	p.mu.Lock()
	defer p.mu.Unlock()
	f()
}

func (p *Player) startSong(s Song) {
	p.out.Start(s)
	for _, l := range p.listeners {
		l.SongChanged(s)
	}
}

func (p *Player) advance(s Song, ok bool) {
	if !ok {
		p.setState(p.state.stop(p))
		return
	}
	p.startSong(s)
	p.setState(playing{})
}

func (p *Player) setState(next state) {
	if next == p.state {
		return
	}
	p.state = next
	for _, l := range p.listeners {
		l.StateChanged(next.name())
	}
}

import random
import threading
from abc import ABC, abstractmethod
from dataclasses import dataclass
from enum import Enum
from typing import Dict, List, Optional


class RepeatMode(Enum):
    OFF = "off"
    ONE = "one"
    ALL = "all"


@dataclass(frozen=True)
class Song:
    id: str
    title: str
    artist: str
    duration_sec: int


class DeviceType(Enum):
    SPEAKER = "speaker"
    WIRED = "wired"
    BLUETOOTH = "bluetooth"
    CAST = "cast"


@dataclass(frozen=True)
class AudioDevice:
    id: str
    name: str
    type: DeviceType


class PlayerListener(ABC):
    @abstractmethod
    def on_song_changed(self, song: Song) -> None: ...

    @abstractmethod
    def on_state_changed(self, state: str) -> None: ...

    @abstractmethod
    def on_device_changed(self, device: AudioDevice) -> None: ...


class AudioOutput(ABC):
    """Adapter target: the player only ever talks to this."""

    @abstractmethod
    def start(self, song: Song) -> None: ...  # from 0:00, playing

    @abstractmethod
    def prepare(self, song: Song, at_sec: int) -> None: ...  # load at a position, paused

    @abstractmethod
    def pause(self) -> None: ...

    @abstractmethod
    def resume(self) -> None: ...

    @abstractmethod
    def stop(self) -> None: ...  # must be safe on a device that has gone away

    @abstractmethod
    def position_sec(self) -> int: ...  # last known position if the device is gone


class CastSession(ABC):
    """A vendor SDK we can't change: URLs, milliseconds, a paused flag."""

    @abstractmethod
    def load_media(self, url: str, start_ms: int, autoplay: bool) -> None: ...

    @abstractmethod
    def set_paused(self, paused: bool) -> None: ...

    @abstractmethod
    def stop_media(self) -> None: ...

    @abstractmethod
    def current_time_ms(self) -> int: ...


class CastOutput(AudioOutput):
    """Adapter: AudioOutput -> CastSession. Bluetooth and speaker adapters look the same."""

    def __init__(self, session: CastSession):
        self._session = session

    @staticmethod
    def _url(song: Song) -> str:
        return "https://cdn.example.com/tracks/" + song.id

    def start(self, song):
        self._session.load_media(self._url(song), 0, True)

    def prepare(self, song, at_sec):
        self._session.load_media(self._url(song), at_sec * 1000, False)

    def pause(self):
        self._session.set_paused(True)

    def resume(self):
        self._session.set_paused(False)

    def stop(self):
        self._session.stop_media()

    def position_sec(self):
        return self._session.current_time_ms() // 1000


class DeviceManager:
    """Connected devices and their adapters. Only Player calls it, under Player's lock."""

    def __init__(self, speaker: AudioDevice, speaker_out: AudioOutput):
        self._devices: Dict[str, AudioDevice] = {}  # insertion order = display order
        self._outputs: Dict[str, AudioOutput] = {}
        self.speaker_id = self.active_id = speaker.id
        self.connected(speaker, speaker_out)

    def connected(self, device: AudioDevice, out: AudioOutput) -> None:
        self._devices[device.id] = device
        self._outputs[device.id] = out

    def disconnected(self, device_id: str) -> None:
        if device_id == self.speaker_id:  # the built-in speaker never goes away
            return
        self._devices.pop(device_id, None)
        self._outputs.pop(device_id, None)

    def available(self) -> List[AudioDevice]:
        return list(self._devices.values())

    def active(self) -> AudioDevice:
        return self._devices[self.active_id]

    def select(self, device_id: str) -> Optional[AudioOutput]:
        out = self._outputs.get(device_id)
        if out is not None:
            self.active_id = device_id
        return out


class PlayQueue:
    def __init__(self, rnd: random.Random):
        self._songs: List[Song] = []
        self._order: List[int] = []  # indices into _songs
        self._pos = -1
        self._rnd = rnd

    def load(self, songs: List[Song], start: int = 0) -> None:
        self._songs = list(songs)
        self._order = list(range(len(songs)))
        self._pos = start if songs else -1

    def current(self) -> Optional[Song]:
        return None if self._pos < 0 else self._songs[self._order[self._pos]]

    def next(self, repeat: RepeatMode, auto: bool) -> Optional[Song]:
        """auto=True when the song ended by itself, False when the user pressed Next."""
        if self._pos < 0:
            return None
        if auto and repeat == RepeatMode.ONE:
            return self.current()
        if self._pos + 1 < len(self._order):
            self._pos += 1
        elif repeat != RepeatMode.OFF:
            self._pos = 0
        else:
            return None
        return self.current()

    def previous(self) -> Optional[Song]:
        if self._pos > 0:
            self._pos -= 1
        return self.current()

    def set_shuffle(self, on: bool) -> None:
        if self._pos < 0:
            return
        cur = self._order[self._pos]
        if not on:
            self._order = list(range(len(self._songs)))
            self._pos = cur
            return
        rest = [i for i in range(len(self._songs)) if i != cur]
        self._rnd.shuffle(rest)  # Fisher-Yates
        self._order = [cur] + rest
        self._pos = 0


class PlayerState(ABC):
    name = ""

    @abstractmethod
    def play(self, p: "Player") -> "PlayerState": ...

    @abstractmethod
    def pause(self, p: "Player") -> "PlayerState": ...

    @abstractmethod
    def stop(self, p: "Player") -> "PlayerState": ...

    @abstractmethod
    def switch_output(self, p: "Player", nxt: AudioOutput) -> "PlayerState": ...


class Stopped(PlayerState):
    name = "STOPPED"

    def play(self, p):
        song = p.queue.current()
        if song is None:
            return self
        p.start_song(song)
        return PLAYING

    def pause(self, p):
        return self

    def stop(self, p):
        return self

    def switch_output(self, p, nxt):
        p.out = nxt
        return self


class Playing(PlayerState):
    name = "PLAYING"

    def play(self, p):
        return self

    def pause(self, p):
        p.out.pause()
        return PAUSED

    def stop(self, p):
        p.out.stop()
        return STOPPED

    def switch_output(self, p, nxt):
        p.move_output(nxt, resume=True)
        return self


class Paused(PlayerState):
    name = "PAUSED"

    def play(self, p):
        p.out.resume()
        return PLAYING

    def pause(self, p):
        return self

    def stop(self, p):
        p.out.stop()
        return STOPPED

    def switch_output(self, p, nxt):
        p.move_output(nxt, resume=False)
        return self


STOPPED, PLAYING, PAUSED = Stopped(), Playing(), Paused()


class Player:
    def __init__(self, queue: PlayQueue, devices: DeviceManager):
        self.queue, self.devices = queue, devices
        self.out: AudioOutput = devices.select(devices.active_id)  # the active device's adapter
        self._listeners: List[PlayerListener] = []
        self._state: PlayerState = STOPPED
        self._repeat = RepeatMode.OFF
        self._lock = threading.RLock()

    def subscribe(self, listener: PlayerListener) -> None:
        with self._lock:
            self._listeners.append(listener)

    def set_repeat(self, mode: RepeatMode) -> None:
        with self._lock:
            self._repeat = mode

    def set_shuffle(self, on: bool) -> None:
        with self._lock:
            self.queue.set_shuffle(on)

    @property
    def state(self) -> str:
        return self._state.name

    def play(self) -> None:
        with self._lock:
            self._set_state(self._state.play(self))

    def pause(self) -> None:
        with self._lock:
            self._set_state(self._state.pause(self))

    def stop(self) -> None:
        with self._lock:
            self._set_state(self._state.stop(self))

    def next(self) -> None:
        with self._lock:
            self._advance(self.queue.next(self._repeat, auto=False))

    def track_ended(self) -> None:
        with self._lock:
            self._advance(self.queue.next(self._repeat, auto=True))

    def previous(self) -> None:
        with self._lock:
            restart = self.out.position_sec() > 3
            self._advance(self.queue.current() if restart else self.queue.previous())

    # ---- device selection ----

    def available_devices(self) -> List[AudioDevice]:
        with self._lock:
            return self.devices.available()

    def device_connected(self, device: AudioDevice, out: AudioOutput) -> None:
        with self._lock:
            self.devices.connected(device, out)

    def select_device(self, device_id: str) -> None:
        with self._lock:
            self._switch_to(device_id)

    def device_disconnected(self, device_id: str) -> None:
        """Called from the OS thread, e.g. Bluetooth dropped. Pause first, then fall back to the speaker."""
        with self._lock:
            if device_id == self.devices.active_id:
                self._set_state(self._state.pause(self))
                self._switch_to(self.devices.speaker_id)
            self.devices.disconnected(device_id)

    def _switch_to(self, device_id: str) -> None:
        nxt = self.devices.select(device_id)
        if nxt is None or nxt is self.out:
            return
        self._set_state(self._state.switch_output(self, nxt))
        device = self.devices.active()
        for listener in self._listeners:
            listener.on_device_changed(device)

    def move_output(self, nxt: AudioOutput, resume: bool) -> None:
        """Same song, same second, on the new device. resume=False keeps it paused."""
        at = self.out.position_sec()
        song = self.queue.current()
        self.out.stop()
        self.out = nxt
        nxt.prepare(song, at)
        if resume:
            nxt.resume()

    def start_song(self, song: Song) -> None:
        self.out.start(song)
        for listener in self._listeners:
            listener.on_song_changed(song)

    def _advance(self, song: Optional[Song]) -> None:
        if song is None:
            self._set_state(self._state.stop(self))
            return
        self.start_song(song)
        self._set_state(PLAYING)

    def _set_state(self, nxt: PlayerState) -> None:
        if nxt is self._state:
            return
        self._state = nxt
        for listener in self._listeners:
            listener.on_state_changed(nxt.name)
