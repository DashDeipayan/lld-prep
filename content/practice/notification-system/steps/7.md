The delivery core a worker runs per dequeued notification. Senders sit in a registry, the idempotency check is atomic, only retryable errors are retried, and `sleep` is injected so tests don't wait.

```java
import java.time.Duration;
import java.util.*;
import java.util.concurrent.ConcurrentHashMap;

enum Channel { EMAIL, SMS, PUSH, IN_APP }

enum Status { SENT, FAILED, SKIPPED, DUPLICATE }

record Notification(String idempotencyKey, String userId, String templateId, String category,
                    Map<String, String> params, List<Channel> channels) {}

record UserPreferences(Map<Channel, String> addresses, Set<Channel> optedOut, Set<String> mutedCategories) {}

record Message(String to, String body) {}

record DeliveryResult(Channel channel, Status status, int attempts, String detail) {}

class SendException extends Exception {
    final boolean retryable;
    SendException(String msg, boolean retryable) { super(msg); this.retryable = retryable; }
}

interface ChannelSender { void send(Message m) throws SendException; }   // Adapter over SES, Twilio, FCM…

interface PreferenceStore { UserPreferences get(String userId); }

interface Sleeper { void sleep(Duration d) throws InterruptedException; }

class TemplateStore {
    private final Map<String, String> templates = new ConcurrentHashMap<>();

    void put(String templateId, Channel c, String text) { templates.put(templateId + ":" + c, text); }

    String render(String templateId, Channel c, Map<String, String> params) {
        String text = templates.get(templateId + ":" + c);
        if (text == null) throw new IllegalArgumentException("no " + c + " template " + templateId);
        for (Map.Entry<String, String> e : params.entrySet()) text = text.replace("{{" + e.getKey() + "}}", e.getValue());
        return text;
    }
}

record RetryPolicy(int maxAttempts, Duration baseDelay) {
    Duration delayBefore(int attempt) { return baseDelay.multipliedBy(1L << (attempt - 2)); }  // exponential
}

class NotificationService {
    private final Map<Channel, ChannelSender> senders;
    private final TemplateStore templates;
    private final PreferenceStore prefs;
    private final RetryPolicy retry;
    private final Sleeper sleeper;
    private final Set<String> seen = ConcurrentHashMap.newKeySet();

    NotificationService(Map<Channel, ChannelSender> senders, TemplateStore templates, PreferenceStore prefs,
                        RetryPolicy retry, Sleeper sleeper) {
        this.senders = Map.copyOf(senders);
        this.templates = templates; this.prefs = prefs; this.retry = retry; this.sleeper = sleeper;
    }

    /** What a worker runs for each dequeued notification. */
    List<DeliveryResult> send(Notification n) {
        List<DeliveryResult> out = new ArrayList<>();
        if (!seen.add(n.idempotencyKey())) {                 // atomic check-and-mark
            for (Channel c : n.channels()) out.add(new DeliveryResult(c, Status.DUPLICATE, 0, "already processed"));
            return out;
        }
        UserPreferences p = prefs.get(n.userId());
        for (Channel c : n.channels()) out.add(deliver(n, c, p));
        return out;
    }

    private DeliveryResult deliver(Notification n, Channel c, UserPreferences p) {
        if (p.optedOut().contains(c) || p.mutedCategories().contains(n.category()))
            return new DeliveryResult(c, Status.SKIPPED, 0, "user preference");
        String to = p.addresses().get(c);
        ChannelSender sender = senders.get(c);
        if (to == null || sender == null) return new DeliveryResult(c, Status.SKIPPED, 0, "no address or sender");
        Message m;
        try {
            m = new Message(to, templates.render(n.templateId(), c, n.params()));
        } catch (IllegalArgumentException e) {
            return new DeliveryResult(c, Status.FAILED, 0, e.getMessage());
        }
        for (int attempt = 1; ; attempt++) {
            try {
                sender.send(m);
                return new DeliveryResult(c, Status.SENT, attempt, null);
            } catch (SendException e) {
                if (!e.retryable || attempt >= retry.maxAttempts())
                    return new DeliveryResult(c, Status.FAILED, attempt, e.getMessage());
            }
            try {
                sleeper.sleep(retry.delayBefore(attempt + 1));
            } catch (InterruptedException ie) {
                Thread.currentThread().interrupt();
                return new DeliveryResult(c, Status.FAILED, attempt, "interrupted");
            }
        }
    }
}
```

```go
package notify

import (
	"errors"
	"strings"
	"sync"
	"time"
)

type Channel string

const (
	Email Channel = "email"
	SMS   Channel = "sms"
	Push  Channel = "push"
	InApp Channel = "in_app"
)

type Status string

const (
	Sent      Status = "sent"
	Failed    Status = "failed"
	Skipped   Status = "skipped"
	Duplicate Status = "duplicate"
)

type Notification struct {
	IdempotencyKey, UserID, TemplateID, Category string
	Params                                       map[string]string
	Channels                                     []Channel
}

type Preferences struct {
	Addresses       map[Channel]string
	OptedOut        map[Channel]bool
	MutedCategories map[string]bool
}

type Message struct{ To, Body string }

type DeliveryResult struct {
	Channel  Channel
	Status   Status
	Attempts int
	Detail   string
}

// Sender is the adapter over a provider: SES, Twilio, FCM…
type Sender interface{ Send(m Message) error }

// SendError tells the service whether a failure is worth retrying.
type SendError struct {
	Msg       string
	Retryable bool
}

func (e *SendError) Error() string { return e.Msg }

type PreferenceStore interface {
	Get(userID string) Preferences
}

type Templates struct {
	mu sync.RWMutex
	t  map[string]string
}

func NewTemplates() *Templates { return &Templates{t: map[string]string{}} }

func (ts *Templates) Put(id string, c Channel, text string) {
	ts.mu.Lock()
	defer ts.mu.Unlock()
	ts.t[id+":"+string(c)] = text
}

func (ts *Templates) Render(id string, c Channel, params map[string]string) (string, error) {
	ts.mu.RLock()
	text, ok := ts.t[id+":"+string(c)]
	ts.mu.RUnlock()
	if !ok {
		return "", errors.New("no " + string(c) + " template " + id)
	}
	for k, v := range params {
		text = strings.ReplaceAll(text, "{{"+k+"}}", v)
	}
	return text, nil
}

type RetryPolicy struct {
	MaxAttempts int
	BaseDelay   time.Duration
}

// DelayBefore returns the exponential backoff before the given attempt (2, 3, …).
func (r RetryPolicy) DelayBefore(attempt int) time.Duration { return r.BaseDelay << (attempt - 2) }

type Service struct {
	senders   map[Channel]Sender
	templates *Templates
	prefs     PreferenceStore
	retry     RetryPolicy
	sleep     func(time.Duration)

	mu   sync.Mutex
	seen map[string]bool
}

func NewService(senders map[Channel]Sender, t *Templates, p PreferenceStore, r RetryPolicy, sleep func(time.Duration)) *Service {
	return &Service{senders: senders, templates: t, prefs: p, retry: r, sleep: sleep, seen: map[string]bool{}}
}

// firstTime atomically checks and marks an idempotency key.
func (s *Service) firstTime(key string) bool {
	s.mu.Lock()
	defer s.mu.Unlock()
	if s.seen[key] {
		return false
	}
	s.seen[key] = true
	return true
}

// Send is what a worker runs for each dequeued notification.
func (s *Service) Send(n Notification) []DeliveryResult {
	out := make([]DeliveryResult, 0, len(n.Channels))
	if !s.firstTime(n.IdempotencyKey) {
		for _, c := range n.Channels {
			out = append(out, DeliveryResult{c, Duplicate, 0, "already processed"})
		}
		return out
	}
	p := s.prefs.Get(n.UserID)
	for _, c := range n.Channels {
		out = append(out, s.deliver(n, c, p))
	}
	return out
}

func (s *Service) deliver(n Notification, c Channel, p Preferences) DeliveryResult {
	if p.OptedOut[c] || p.MutedCategories[n.Category] {
		return DeliveryResult{c, Skipped, 0, "user preference"}
	}
	to, snd := p.Addresses[c], s.senders[c]
	if to == "" || snd == nil {
		return DeliveryResult{c, Skipped, 0, "no address or sender"}
	}
	body, err := s.templates.Render(n.TemplateID, c, n.Params)
	if err != nil {
		return DeliveryResult{c, Failed, 0, err.Error()}
	}
	for attempt := 1; ; attempt++ {
		err := snd.Send(Message{To: to, Body: body})
		if err == nil {
			return DeliveryResult{c, Sent, attempt, ""}
		}
		var se *SendError
		if !errors.As(err, &se) || !se.Retryable || attempt >= s.retry.MaxAttempts {
			return DeliveryResult{c, Failed, attempt, err.Error()}
		}
		s.sleep(s.retry.DelayBefore(attempt + 1))
	}
}
```

```python
import threading
import time
from abc import ABC, abstractmethod
from dataclasses import dataclass
from datetime import timedelta
from enum import Enum
from typing import Callable, Dict, FrozenSet, List, Optional


class Channel(Enum):
    EMAIL = "email"
    SMS = "sms"
    PUSH = "push"
    IN_APP = "in_app"


class Status(Enum):
    SENT = "sent"
    FAILED = "failed"
    SKIPPED = "skipped"
    DUPLICATE = "duplicate"


@dataclass(frozen=True)
class Notification:
    idempotency_key: str
    user_id: str
    template_id: str
    category: str
    params: Dict[str, str]
    channels: List[Channel]


@dataclass(frozen=True)
class UserPreferences:
    addresses: Dict[Channel, str]
    opted_out: FrozenSet[Channel] = frozenset()
    muted_categories: FrozenSet[str] = frozenset()


@dataclass(frozen=True)
class Message:
    to: str
    body: str


@dataclass(frozen=True)
class DeliveryResult:
    channel: Channel
    status: Status
    attempts: int
    detail: Optional[str] = None


class SendError(Exception):
    def __init__(self, msg: str, retryable: bool):
        super().__init__(msg)
        self.retryable = retryable


class ChannelSender(ABC):  # adapter over a provider: SES, Twilio, FCM...
    @abstractmethod
    def send(self, message: Message) -> None: ...


class PreferenceStore(ABC):
    @abstractmethod
    def get(self, user_id: str) -> UserPreferences: ...


class TemplateStore:
    def __init__(self):
        self._templates: Dict[str, str] = {}

    def put(self, template_id: str, channel: Channel, text: str) -> None:
        self._templates["%s:%s" % (template_id, channel.value)] = text

    def render(self, template_id: str, channel: Channel, params: Dict[str, str]) -> str:
        text = self._templates.get("%s:%s" % (template_id, channel.value))
        if text is None:
            raise KeyError("no %s template %s" % (channel.value, template_id))
        for k, v in params.items():
            text = text.replace("{{%s}}" % k, v)
        return text


@dataclass(frozen=True)
class RetryPolicy:
    max_attempts: int
    base_delay: timedelta

    def delay_before(self, attempt: int) -> timedelta:
        return self.base_delay * (2 ** (attempt - 2))  # exponential


class NotificationService:
    def __init__(self, senders: Dict[Channel, ChannelSender], templates: TemplateStore,
                 prefs: PreferenceStore, retry: RetryPolicy,
                 sleep: Callable[[float], None] = time.sleep):
        self._senders = dict(senders)
        self._templates, self._prefs, self._retry, self._sleep = templates, prefs, retry, sleep
        self._seen = set()
        self._lock = threading.Lock()

    def _first_time(self, key: str) -> bool:
        with self._lock:  # atomic check-and-mark
            if key in self._seen:
                return False
            self._seen.add(key)
            return True

    def send(self, n: Notification) -> List[DeliveryResult]:
        """What a worker runs for each dequeued notification."""
        if not self._first_time(n.idempotency_key):
            return [DeliveryResult(c, Status.DUPLICATE, 0, "already processed") for c in n.channels]
        prefs = self._prefs.get(n.user_id)
        return [self._deliver(n, c, prefs) for c in n.channels]

    def _deliver(self, n: Notification, c: Channel, p: UserPreferences) -> DeliveryResult:
        if c in p.opted_out or n.category in p.muted_categories:
            return DeliveryResult(c, Status.SKIPPED, 0, "user preference")
        to, sender = p.addresses.get(c), self._senders.get(c)
        if to is None or sender is None:
            return DeliveryResult(c, Status.SKIPPED, 0, "no address or sender")
        try:
            message = Message(to, self._templates.render(n.template_id, c, n.params))
        except KeyError as e:
            return DeliveryResult(c, Status.FAILED, 0, str(e))
        attempt = 1
        while True:
            try:
                sender.send(message)
                return DeliveryResult(c, Status.SENT, attempt)
            except SendError as e:
                if not e.retryable or attempt >= self._retry.max_attempts:
                    return DeliveryResult(c, Status.FAILED, attempt, str(e))
            attempt += 1
            self._sleep(self._retry.delay_before(attempt).total_seconds())
```

## Compare with yours

- Is there **no `switch` on channel**? Senders should come from a map.
- Are preferences checked **before** rendering and sending?
- Do you retry **only retryable** errors, with growing delays?
- Is the idempotency check **one atomic operation** (`Set.add` / `putIfAbsent` / mutex)?
- In production, retries become **delayed re-enqueues** rather than `sleep`, and the idempotency mark should be paired with stored results so a retried request gets the original outcome. Mention both.
