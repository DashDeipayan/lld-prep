```java
import java.time.*;
import java.util.*;
import java.util.concurrent.atomic.AtomicLong;

enum SpotSize { SMALL, MEDIUM, LARGE }

enum VehicleType {
    BIKE(SpotSize.SMALL), CAR(SpotSize.MEDIUM), TRUCK(SpotSize.LARGE);
    final SpotSize minSize;
    VehicleType(SpotSize s) { minSize = s; }
}

record Vehicle(String plate, VehicleType type) {}
record Ticket(String id, String plate, VehicleType type, String spotId, Instant entry) {}
record Receipt(Ticket ticket, Instant exit, long fee) {}

interface PricingStrategy { long fee(VehicleType type, Duration parked); }
interface SpotSelector { Spot select(List<Spot> free, Vehicle v); }

class Spot {
    final String id; final int floor; final SpotSize size;
    private Vehicle vehicle;
    Spot(String id, int floor, SpotSize size) { this.id = id; this.floor = floor; this.size = size; }
    boolean isFree() { return vehicle == null; }
    boolean canFit(Vehicle v) { return size.compareTo(v.type().minSize) >= 0; }
    void occupy(Vehicle v) {
        if (!isFree() || !canFit(v)) throw new IllegalStateException("spot " + id + " unavailable");
        vehicle = v;
    }
    Vehicle vacate() { Vehicle v = vehicle; vehicle = null; return v; }
}

class HourlyPricing implements PricingStrategy {
    private final Map<VehicleType, Long> rate;
    HourlyPricing(Map<VehicleType, Long> rate) { this.rate = rate; }
    public long fee(VehicleType t, Duration d) {
        long seconds = Math.max(0, d.getSeconds());
        long hours = Math.max(1, (seconds + 3599) / 3600);   // started hours, min 1
        return hours * rate.get(t);
    }
}

class NearestSpotSelector implements SpotSelector {
    public Spot select(List<Spot> free, Vehicle v) { return free.isEmpty() ? null : free.get(0); }
}

class ParkingLot {
    private final List<Spot> spots = new ArrayList<>();          // lot order
    private final Map<String, Spot> byId = new HashMap<>();
    private final Map<String, Ticket> active = new HashMap<>();
    private final Set<String> platesInside = new HashSet<>();
    private final Set<Integer> floors = new HashSet<>();
    private final PricingStrategy pricing;
    private final SpotSelector selector;
    private final Clock clock;
    private final AtomicLong seq = new AtomicLong();

    ParkingLot(PricingStrategy pricing, SpotSelector selector, Clock clock) {
        this.pricing = pricing; this.selector = selector; this.clock = clock;
    }

    synchronized void addFloor(int floor, int small, int medium, int large) {
        if (!floors.add(floor)) throw new IllegalArgumentException("duplicate floor " + floor);
        add(floor, SpotSize.SMALL, "S", small);
        add(floor, SpotSize.MEDIUM, "M", medium);
        add(floor, SpotSize.LARGE, "L", large);
    }

    private void add(int floor, SpotSize size, String code, int n) {
        for (int i = 1; i <= n; i++) {
            Spot s = new Spot("F" + floor + "-" + code + i, floor, size);
            spots.add(s);
            byId.put(s.id, s);
        }
    }

    synchronized Ticket park(Vehicle v) {
        if (platesInside.contains(v.plate())) throw new IllegalStateException("already parked: " + v.plate());
        List<Spot> free = new ArrayList<>();
        for (Spot s : spots) if (s.isFree() && s.canFit(v)) free.add(s);
        Spot spot = selector.select(free, v);
        if (spot == null) throw new IllegalStateException("no spot for " + v.type());
        spot.occupy(v);
        Ticket t = new Ticket("T-" + seq.incrementAndGet(), v.plate(), v.type(), spot.id, clock.instant());
        active.put(t.id(), t);
        platesInside.add(v.plate());
        return t;
    }

    synchronized Receipt unpark(String ticketId) {
        Ticket t = active.remove(ticketId);
        if (t == null) throw new IllegalArgumentException("unknown ticket " + ticketId);
        byId.get(t.spotId()).vacate();
        platesInside.remove(t.plate());
        Instant exit = clock.instant();
        return new Receipt(t, exit, pricing.fee(t.type(), Duration.between(t.entry(), exit)));
    }

    synchronized int available(SpotSize size) {
        int n = 0;
        for (Spot s : spots) if (s.size == size && s.isFree()) n++;
        return n;
    }
}
```

```go
package parking

import (
	"errors"
	"fmt"
	"sync"
	"time"
)

type SpotSize int

const (
	Small SpotSize = iota
	Medium
	Large
)

type VehicleType int

const (
	Bike VehicleType = iota
	Car
	Truck
)

func (t VehicleType) MinSize() SpotSize { return [...]SpotSize{Small, Medium, Large}[t] }

type Vehicle struct {
	Plate string
	Type  VehicleType
}

type Ticket struct {
	ID, Plate, SpotID string
	Type              VehicleType
	Entry             time.Time
}

type Receipt struct {
	Ticket Ticket
	Exit   time.Time
	Fee    int64
}

type PricingStrategy interface {
	Fee(t VehicleType, parked time.Duration) int64
}

type SpotSelector interface {
	Select(free []*Spot, v Vehicle) *Spot
}

var (
	ErrAlreadyParked  = errors.New("vehicle already parked")
	ErrLotFull        = errors.New("no spot fits this vehicle")
	ErrUnknownTicket  = errors.New("unknown ticket")
	ErrDuplicateFloor = errors.New("duplicate floor")
)

type Spot struct {
	ID      string
	Floor   int
	Size    SpotSize
	vehicle *Vehicle
}

func (s *Spot) Free() bool            { return s.vehicle == nil }
func (s *Spot) CanFit(v Vehicle) bool { return s.Size >= v.Type.MinSize() }

type HourlyPricing map[VehicleType]int64

func (p HourlyPricing) Fee(t VehicleType, d time.Duration) int64 {
	hours := int64((d + time.Hour - 1) / time.Hour) // started hours
	if hours < 1 {
		hours = 1
	}
	return hours * p[t]
}

type NearestSpotSelector struct{}

func (NearestSpotSelector) Select(free []*Spot, _ Vehicle) *Spot {
	if len(free) == 0 {
		return nil
	}
	return free[0]
}

type Lot struct {
	mu       sync.Mutex
	spots    []*Spot // lot order
	byID     map[string]*Spot
	active   map[string]Ticket
	inside   map[string]bool
	floors   map[int]bool
	pricing  PricingStrategy
	selector SpotSelector
	now      func() time.Time
	seq      int
}

func NewLot(p PricingStrategy, s SpotSelector, now func() time.Time) *Lot {
	return &Lot{byID: map[string]*Spot{}, active: map[string]Ticket{}, inside: map[string]bool{},
		floors: map[int]bool{}, pricing: p, selector: s, now: now}
}

func (l *Lot) AddFloor(floor, small, medium, large int) error {
	l.mu.Lock()
	defer l.mu.Unlock()
	if l.floors[floor] {
		return ErrDuplicateFloor
	}
	l.floors[floor] = true
	for _, g := range []struct {
		size SpotSize
		code string
		n    int
	}{{Small, "S", small}, {Medium, "M", medium}, {Large, "L", large}} {
		for i := 1; i <= g.n; i++ {
			s := &Spot{ID: fmt.Sprintf("F%d-%s%d", floor, g.code, i), Floor: floor, Size: g.size}
			l.spots = append(l.spots, s)
			l.byID[s.ID] = s
		}
	}
	return nil
}

func (l *Lot) Park(v Vehicle) (Ticket, error) {
	l.mu.Lock()
	defer l.mu.Unlock()
	if l.inside[v.Plate] {
		return Ticket{}, ErrAlreadyParked
	}
	var free []*Spot
	for _, s := range l.spots {
		if s.Free() && s.CanFit(v) {
			free = append(free, s)
		}
	}
	spot := l.selector.Select(free, v)
	if spot == nil {
		return Ticket{}, ErrLotFull
	}
	spot.vehicle = &v
	l.seq++
	t := Ticket{ID: fmt.Sprintf("T-%d", l.seq), Plate: v.Plate, SpotID: spot.ID, Type: v.Type, Entry: l.now()}
	l.active[t.ID] = t
	l.inside[v.Plate] = true
	return t, nil
}

func (l *Lot) Unpark(ticketID string) (Receipt, error) {
	l.mu.Lock()
	defer l.mu.Unlock()
	t, ok := l.active[ticketID]
	if !ok {
		return Receipt{}, ErrUnknownTicket
	}
	delete(l.active, ticketID)
	l.byID[t.SpotID].vehicle = nil
	delete(l.inside, t.Plate)
	exit := l.now()
	return Receipt{Ticket: t, Exit: exit, Fee: l.pricing.Fee(t.Type, exit.Sub(t.Entry))}, nil
}

func (l *Lot) Available(size SpotSize) int {
	l.mu.Lock()
	defer l.mu.Unlock()
	n := 0
	for _, s := range l.spots {
		if s.Size == size && s.Free() {
			n++
		}
	}
	return n
}
```

```python
import itertools
import math
import threading
from abc import ABC, abstractmethod
from dataclasses import dataclass
from datetime import datetime, timedelta
from enum import Enum, IntEnum
from typing import Callable, Dict, List, Optional


class SpotSize(IntEnum):
    SMALL = 0
    MEDIUM = 1
    LARGE = 2


class VehicleType(Enum):
    BIKE = SpotSize.SMALL
    CAR = SpotSize.MEDIUM
    TRUCK = SpotSize.LARGE

    @property
    def min_size(self) -> SpotSize:
        return self.value


@dataclass(frozen=True)
class Vehicle:
    plate: str
    type: VehicleType


@dataclass(frozen=True)
class Ticket:
    id: str
    plate: str
    type: VehicleType
    spot_id: str
    entry: datetime


@dataclass(frozen=True)
class Receipt:
    ticket: Ticket
    exit: datetime
    fee: int


class PricingStrategy(ABC):
    @abstractmethod
    def fee(self, vtype: VehicleType, parked: timedelta) -> int: ...


class SpotSelector(ABC):
    @abstractmethod
    def select(self, free: List["Spot"], vehicle: Vehicle) -> Optional["Spot"]: ...


class Spot:
    def __init__(self, spot_id: str, floor: int, size: SpotSize):
        self.id, self.floor, self.size = spot_id, floor, size
        self.vehicle: Optional[Vehicle] = None

    def is_free(self) -> bool:
        return self.vehicle is None

    def can_fit(self, v: Vehicle) -> bool:
        return self.size >= v.type.min_size


class HourlyPricing(PricingStrategy):
    def __init__(self, rates: Dict[VehicleType, int]):
        self._rates = rates

    def fee(self, vtype: VehicleType, parked: timedelta) -> int:
        hours = max(1, math.ceil(parked.total_seconds() / 3600))
        return hours * self._rates[vtype]


class NearestSpotSelector(SpotSelector):
    def select(self, free: List[Spot], vehicle: Vehicle) -> Optional[Spot]:
        return free[0] if free else None


class ParkingLot:
    def __init__(self, pricing: PricingStrategy, selector: SpotSelector,
                 now: Callable[[], datetime] = datetime.now):
        self._spots: List[Spot] = []          # lot order
        self._by_id: Dict[str, Spot] = {}
        self._active: Dict[str, Ticket] = {}
        self._inside = set()
        self._floors = set()
        self._pricing, self._selector, self._now = pricing, selector, now
        self._seq = itertools.count(1)
        self._lock = threading.Lock()

    def add_floor(self, floor: int, small: int, medium: int, large: int) -> None:
        with self._lock:
            if floor in self._floors:
                raise ValueError("duplicate floor %d" % floor)
            self._floors.add(floor)
            for size, code, n in ((SpotSize.SMALL, "S", small), (SpotSize.MEDIUM, "M", medium),
                                  (SpotSize.LARGE, "L", large)):
                for i in range(1, n + 1):
                    s = Spot("F%d-%s%d" % (floor, code, i), floor, size)
                    self._spots.append(s)
                    self._by_id[s.id] = s

    def park(self, v: Vehicle) -> Ticket:
        with self._lock:
            if v.plate in self._inside:
                raise ValueError("already parked: " + v.plate)
            free = [s for s in self._spots if s.is_free() and s.can_fit(v)]
            spot = self._selector.select(free, v)
            if spot is None:
                raise ValueError("no spot for " + v.type.name)
            spot.vehicle = v
            t = Ticket("T-%d" % next(self._seq), v.plate, v.type, spot.id, self._now())
            self._active[t.id] = t
            self._inside.add(v.plate)
            return t

    def unpark(self, ticket_id: str) -> Receipt:
        with self._lock:
            t = self._active.pop(ticket_id, None)
            if t is None:
                raise KeyError("unknown ticket " + ticket_id)
            self._by_id[t.spot_id].vehicle = None
            self._inside.discard(t.plate)
            exit_time = self._now()
            return Receipt(t, exit_time, self._pricing.fee(t.type, exit_time - t.entry))

    def available(self, size: SpotSize) -> int:
        with self._lock:
            return sum(1 for s in self._spots if s.size == size and s.is_free())
```
