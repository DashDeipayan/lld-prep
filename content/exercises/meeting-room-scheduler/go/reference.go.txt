package main

import (
	"errors"
	"fmt"
	"slices"
	"sort"
	"sync"
)

// ======================= Provided =======================

type Booking struct {
	ID, Room, Organizer string
	Start, End          int
}

type Slot struct{ Start, End int }

var (
	ErrInvalid  = errors.New("invalid request")
	ErrNotFound = errors.New("not found")
	ErrConflict = errors.New("conflict")
	ErrNoRoom   = errors.New("no room available")
)

// ======================= Model =======================

type room struct {
	id       string
	capacity int
	bookings []Booking // sorted by start
}

func (r *room) free(s, e int) bool {
	for _, b := range r.bookings {
		if b.Start >= e {
			break
		}
		if s < b.End && b.Start < e {
			return false
		}
	}
	return true
}

func (r *room) insert(b Booking) {
	i := sort.Search(len(r.bookings), func(i int) bool { return r.bookings[i].Start >= b.Start })
	r.bookings = slices.Insert(r.bookings, i, b)
}

type Scheduler struct {
	mu       sync.Mutex
	rooms    map[string]*room
	bookings map[string]Booking
	next     int
}

func NewScheduler() *Scheduler {
	return &Scheduler{rooms: map[string]*room{}, bookings: map[string]Booking{}}
}

func valid(start, end int) error {
	if start < 0 || end > 1440 || start >= end {
		return fmt.Errorf("%w: interval [%d,%d)", ErrInvalid, start, end)
	}
	return nil
}

func (s *Scheduler) AddRoom(id string, capacity int) error {
	s.mu.Lock()
	defer s.mu.Unlock()
	if _, dup := s.rooms[id]; dup || capacity < 1 {
		return fmt.Errorf("%w: room %s", ErrInvalid, id)
	}
	s.rooms[id] = &room{id: id, capacity: capacity}
	return nil
}

// bestFit orders rooms by capacity, then id.
func (s *Scheduler) candidates(start, end, attendees int) []*room {
	var out []*room
	for _, r := range s.rooms {
		if r.capacity >= attendees && r.free(start, end) {
			out = append(out, r)
		}
	}
	sort.Slice(out, func(i, j int) bool {
		if out[i].capacity != out[j].capacity {
			return out[i].capacity < out[j].capacity
		}
		return out[i].id < out[j].id
	})
	return out
}

func (s *Scheduler) place(r *room, organizer string, start, end int) Booking {
	s.next++
	b := Booking{ID: fmt.Sprintf("M%d", s.next), Room: r.id, Organizer: organizer, Start: start, End: end}
	r.insert(b)
	s.bookings[b.ID] = b
	return b
}

func (s *Scheduler) Book(organizer string, start, end, attendees int) (Booking, error) {
	if err := valid(start, end); err != nil {
		return Booking{}, err
	}
	if attendees < 1 {
		return Booking{}, fmt.Errorf("%w: attendees", ErrInvalid)
	}
	s.mu.Lock()
	defer s.mu.Unlock()
	c := s.candidates(start, end, attendees)
	if len(c) == 0 {
		return Booking{}, ErrNoRoom
	}
	return s.place(c[0], organizer, start, end), nil
}

func (s *Scheduler) BookRoom(id, organizer string, start, end int) (Booking, error) {
	if err := valid(start, end); err != nil {
		return Booking{}, err
	}
	s.mu.Lock()
	defer s.mu.Unlock()
	r, ok := s.rooms[id]
	if !ok {
		return Booking{}, fmt.Errorf("%w: room %s", ErrNotFound, id)
	}
	if !r.free(start, end) {
		return Booking{}, fmt.Errorf("%w: %s", ErrConflict, id)
	}
	return s.place(r, organizer, start, end), nil
}

func (s *Scheduler) Cancel(id string) error {
	s.mu.Lock()
	defer s.mu.Unlock()
	b, ok := s.bookings[id]
	if !ok {
		return fmt.Errorf("%w: booking %s", ErrNotFound, id)
	}
	delete(s.bookings, id)
	r := s.rooms[b.Room]
	r.bookings = slices.DeleteFunc(r.bookings, func(x Booking) bool { return x.ID == id })
	return nil
}

func (s *Scheduler) FreeRooms(start, end, attendees int) ([]string, error) {
	if err := valid(start, end); err != nil {
		return nil, err
	}
	s.mu.Lock()
	defer s.mu.Unlock()
	out := []string{}
	for _, r := range s.candidates(start, end, attendees) {
		out = append(out, r.id)
	}
	return out, nil
}

func (s *Scheduler) Schedule(id string) ([]Booking, error) {
	s.mu.Lock()
	defer s.mu.Unlock()
	r, ok := s.rooms[id]
	if !ok {
		return nil, fmt.Errorf("%w: room %s", ErrNotFound, id)
	}
	return append([]Booking{}, r.bookings...), nil
}

func (s *Scheduler) FreeSlots(id string, from, to int) ([]Slot, error) {
	if err := valid(from, to); err != nil {
		return nil, err
	}
	s.mu.Lock()
	defer s.mu.Unlock()
	r, ok := s.rooms[id]
	if !ok {
		return nil, fmt.Errorf("%w: room %s", ErrNotFound, id)
	}
	out := []Slot{}
	cursor := from
	for _, b := range r.bookings {
		if b.End <= cursor {
			continue
		}
		if b.Start >= to {
			break
		}
		if b.Start > cursor {
			out = append(out, Slot{cursor, b.Start})
		}
		cursor = max(cursor, b.End)
	}
	if cursor < to {
		out = append(out, Slot{cursor, to})
	}
	return out, nil
}
