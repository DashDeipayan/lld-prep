package main

import "sync"

type node[K comparable, V any] struct {
	key        K
	value      V
	prev, next *node[K, V]
}

type LRU[K comparable, V any] struct {
	mu       sync.Mutex
	capacity int
	onEvict  func(K, V)
	items    map[K]*node[K, V]
	head     *node[K, V] // sentinel: head.next is MRU
	tail     *node[K, V] // sentinel: tail.prev is LRU
}

func NewLRU[K comparable, V any](capacity int, onEvict func(K, V)) *LRU[K, V] {
	if capacity < 1 {
		panic("capacity must be >= 1")
	}
	c := &LRU[K, V]{capacity: capacity, onEvict: onEvict, items: map[K]*node[K, V]{},
		head: &node[K, V]{}, tail: &node[K, V]{}}
	c.head.next, c.tail.prev = c.tail, c.head
	return c
}

func (c *LRU[K, V]) unlink(n *node[K, V]) {
	n.prev.next, n.next.prev = n.next, n.prev
}

func (c *LRU[K, V]) pushFront(n *node[K, V]) {
	n.prev, n.next = c.head, c.head.next
	c.head.next.prev = n
	c.head.next = n
}

func (c *LRU[K, V]) Get(k K) (V, bool) {
	c.mu.Lock()
	defer c.mu.Unlock()
	n, ok := c.items[k]
	if !ok {
		var zero V
		return zero, false
	}
	c.unlink(n)
	c.pushFront(n)
	return n.value, true
}

func (c *LRU[K, V]) Put(k K, v V) {
	var evicted *node[K, V]
	c.mu.Lock()
	if n, ok := c.items[k]; ok {
		n.value = v
		c.unlink(n)
		c.pushFront(n)
		c.mu.Unlock()
		return
	}
	n := &node[K, V]{key: k, value: v}
	c.items[k] = n
	c.pushFront(n)
	if len(c.items) > c.capacity {
		evicted = c.tail.prev
		c.unlink(evicted)
		delete(c.items, evicted.key)
	}
	c.mu.Unlock()
	// Callback outside the lock so user code can't deadlock us.
	if evicted != nil && c.onEvict != nil {
		c.onEvict(evicted.key, evicted.value)
	}
}

func (c *LRU[K, V]) Len() int {
	c.mu.Lock()
	defer c.mu.Unlock()
	return len(c.items)
}

func (c *LRU[K, V]) Keys() []K {
	c.mu.Lock()
	defer c.mu.Unlock()
	out := make([]K, 0, len(c.items))
	for n := c.head.next; n != c.tail; n = n.next {
		out = append(out, n.key)
	}
	return out
}
